\def\TCID{TC-0019}
\def\TCTitle{%
  Engine entry points, registration, contexts and projects}
\def\TCPurpose{%
  Verify \texttt{DllFunctionCall} (error numbers and texts measured with a
  Visual Basic 6 program), \texttt{End}, \texttt{Stop}, \texttt{ThunRTMain},
  the p-code entries (in child processes, since they end the program), the
  ActiveX DLL registration under a private
  \texttt{HKEY\_\allowbreak{}CLASSES\_\allowbreak{}ROOT}, contexts,
  projects, the type-information provider calls and the help hooks.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0002, REQ-0003}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}engine\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}engine.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing; child processes isolate the entries that end
  the program.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest. Memory below 4 GB is available for a project
  header. The registration checks do not run under Wine.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  Declare'd calls with bad arguments, project headers built by the test, and
  the test's own executable run as a child.}
\def\TCInitialState{%
  The working directory \texttt{work-engine}, used by no other test.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}engine\_\allowbreak{}tests}.
    \item Check \texttt{DllFunctionCall} error numbers and texts, contexts and
      the type-information provider.
    \item Register and unregister an ActiveX DLL under a private registry root
      (not under Wine).
    \item Run the program-ending entries in child processes and check their exit
      codes and diagnostics.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{vbrun engine: all checks
  passed} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}

\def\TCID{TC-0018}
\def\TCTitle{%
  Get \# and Put \# of records and arrays; Print on an object}
\def\TCPurpose{%
  Verify \texttt{\_\allowbreak{}\_\allowbreak{}vbaGetOwner3}/\texttt{4},
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaPutOwner3}/\texttt{4} and
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaPrintObj}, against file bytes
  measured with Visual Basic 6 programs.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}owner\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}openfiles.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Back-to-back comparison with measured file bytes.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  Records and arrays laid out as the compiler lays them out, with this
  target's padding.}
\def\TCInitialState{%
  The working directory \texttt{work-owner}, used by no other test.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}owner\_\allowbreak{}tests}.
    \item Write records and arrays with the Put helpers and read the file back.
    \item Read them with the Get helpers.
    \item Print an object with
      \texttt{\_\allowbreak{}\_\allowbreak{}vbaPrintObj}.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  The file bytes equal the measured bytes; the program prints \texttt{vbrun
  owner and print: all checks passed} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\def\TCID{TC-0013}
\def\TCTitle{%
  The x87 conversion helpers}
\def\TCPurpose{%
  Verify the conversion helpers that work on the x87 stack: the portable
  functions on every architecture, and on 32-bit x86 the reference's own x87
  sequences compared with them value by value, with the floating-point
  exception flags.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0006}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}x87\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}x87.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Back-to-back comparison of the two forms; boundary values of each integer
  and floating-point type.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  Integers, floating-point values, Currency values and values that overflow
  the target type.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}x87\_\allowbreak{}tests}.
    \item Convert the values with the portable functions and check the results
      and error numbers.
    \item On 32-bit x86, run the x87 sequences and compare them with the
      portable results.
    \item Check that a conversion error clears the exception flags.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{vbrun x87: all checks passed}
  and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}

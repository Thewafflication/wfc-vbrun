\def\TCID{TC-0007}
\def\TCTitle{%
  Late-bound calls through IDispatch}
\def\TCPurpose{%
  Verify the late-binding helpers against a small \texttt{IDispatch}
  implemented by the test.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}late\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}late.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing with a test double.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The test's \texttt{IDispatch} object; calls, property reads and
  assignments, \texttt{Set}, by name and by id, with named arguments, on an
  object and on a Variant, and \texttt{CallByName}; an unknown member, an
  unknown argument name, a non-object and no object.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}late\_\allowbreak{}tests}.
    \item Call the late-binding helpers on the test's \texttt{IDispatch} in each
      form (with a result, as a statement, property read and assignment,
      \texttt{Set}, by name and by id, named arguments, on a Variant,
      \texttt{CallByName}).
    \item Check the call flags and argument count, the values returned and the
      errors raised: 424 for \texttt{Set} of a non-object, 438 for an
      unknown member, 91 for no object, 448 for an unknown argument name.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{late ok} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}

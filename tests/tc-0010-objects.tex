\def\TCID{TC-0010}
\def\TCTitle{%
  CreateObject and GetObject}
\def\TCPurpose{%
  Verify \texttt{CreateObject}, \texttt{GetObject} and \texttt{New}
  (\texttt{rtcCreateObject2}, \texttt{rtcCreateObject},
  \texttt{rtcGetObject}, \texttt{\_\allowbreak{}\_\allowbreak{}vbaNew},
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaNew2}).}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}objects\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}objects.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  \texttt{New} of a class, the older \texttt{CreateObject}, the class
  \texttt{Scripting.\allowbreak{}Dictionary}, an unknown class, an empty
  class name, and the path of a file that does not exist.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}objects\_\allowbreak{}tests}.
    \item Create objects of a known class and of unknown classes.
    \item Call \texttt{GetObject} with nothing to get and with a missing file.
    \item Check the objects returned and that an unknown or empty class and a
      missing file raise error 429.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{objects ok} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}

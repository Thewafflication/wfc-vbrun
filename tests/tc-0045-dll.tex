\def\TCID{TC-0045}
\def\TCTitle{%
  DLL black-box check: exports, version resource and registration}
\def\TCPurpose{%
  Verify, with a C program, that \texttt{wfcvbrun.\allowbreak{}dll} loads,
  exports the four COM entry points, carries the version resource of the
  library and target it was built from, and completes a registration round
  trip in the registry of the current user.}
\def\TCPriority{\TCPriorityCritical}

\def\TCRequirementRef{%
  REQ-0001, REQ-0002}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}dll\_\allowbreak{}test.\allowbreak{}c},
  \texttt{tests/\allowbreak{}dll\_\allowbreak{}check.\allowbreak{}h},
  \texttt{version.\allowbreak{}rc.\allowbreak{}in}}
\def\TCTestTechnique{%
  Black-box testing of the binary interface.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest. The registry of the current user holds no
  registration of the library (otherwise the test skips with exit code 77).}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The DLL path, the version
  (\texttt{WFC\_\allowbreak{}VBRUN\_\allowbreak{}VERSION}) and the target
  (\texttt{WFC\_\allowbreak{}LIB\_\allowbreak{}TARGET}) given on the command
  line.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}dll\_\allowbreak{}test
      \textless{}dll\textgreater{} \textless{}version\textgreater{}
      \textless{}target\textgreater{}}.
    \item Load the DLL and look up \texttt{DllCanUnloadNow},
      \texttt{DllGetClassObject}, \texttt{DllRegisterServer} and
      \texttt{DllUnregisterServer}.
    \item Compare the fixed file version and the Comments string of the version
      resource with the version and target.
    \item Call \texttt{DllRegisterServer}, check the library key and server
      value in the registry of the current user, then call
      \texttt{DllUnregisterServer} and check that the keys are removed.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass and the program exits 0. \texttt{DllCanUnloadNow} answers
  \texttt{S\_\allowbreak{}OK}; the library registers no COM classes. Exit
  code 77 means the registration already existed, and the test is skipped.}
\def\TCPostConditions{%
  The registry holds no key of the library; the test holds the CTest
  resource lock \texttt{registry}.}

\TCTemplate

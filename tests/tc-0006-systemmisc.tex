\def\TCID{TC-0006}
\def\TCTitle{%
  StrComp, Command\$ and the small host exports}
\def\TCPurpose{%
  Verify \texttt{StrComp} on Variants, \texttt{Command\$}, \texttt{Chr} and
  the small host exports (type checks, array rebase, move, copy, lock and
  destroy, Variant to bytes, String arrays to ANSI and Unicode,
  \texttt{ChDrive}, the \texttt{Error} statement, the free-list helpers)
  with the results measured from a Visual Basic 6 program.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}system\_\allowbreak{}misc\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}system.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}scalars.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing with measured reference values.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The arguments of the experiment \texttt{cmdLine} and the comparison modes
  of \texttt{StrComp}.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs
      \texttt{vbrun\_\allowbreak{}system\_\allowbreak{}misc\_\allowbreak{}tests}.
    \item Call the exports with the measured arguments.
    \item Compare each result and error with the measured value.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  Every check passes; the program prints \texttt{system misc ok} and exits
  0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

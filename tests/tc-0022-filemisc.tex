\def\TCID{TC-0022}
\def\TCTitle{%
  Width, FileAttr, Name, Lock, Input\$ and fixed-length Get and Put}
\def\TCPurpose{%
  Verify \texttt{Width}, \texttt{FileAttr}, \texttt{Name}, \texttt{Lock} and
  \texttt{Unlock}, \texttt{Input\$} and \texttt{InputB\$}, and the
  fixed-length String forms of \texttt{Get} and \texttt{Put}.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}file\_\allowbreak{}misc\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}openfiles.\allowbreak{}cpp},
  \texttt{tests/\allowbreak{}cases/\allowbreak{}filemisc.\allowbreak{}expected}}
\def\TCTestTechnique{%
  Back-to-back comparison with recorded reference output.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The statements of the experiments \texttt{fileMisc} and \texttt{fileCtrl}
  of
  \texttt{scripts/\allowbreak{}vb6\_\allowbreak{}harness\_\allowbreak{}gen.\allowbreak{}py}.}
\def\TCInitialState{%
  The working directory \texttt{work-filemisc}, used by no other test.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs
      \texttt{vbrun\_\allowbreak{}file\_\allowbreak{}misc\_\allowbreak{}tests
      tests/\allowbreak{}cases/\allowbreak{}filemisc.\allowbreak{}expected}.
    \item Run the statement groups through the helpers.
    \item Record each result and the hexadecimal contents of the files made.
    \item Compare the lines with \texttt{filemisc.\allowbreak{}expected}.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  Every recorded line equals the reference's; the program exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

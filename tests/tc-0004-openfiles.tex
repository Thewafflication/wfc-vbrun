\def\TCID{TC-0004}
\def\TCTitle{%
  Open-file table statements}
\def\TCPurpose{%
  Verify the open-file table (\texttt{Open}, \texttt{Close},
  \texttt{FreeFile}, \texttt{LOF}, \texttt{Loc}, \texttt{Seek},
  \texttt{EOF}, \texttt{Line Input}) against the results of a Visual Basic 6
  program.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}file\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}openfiles.\allowbreak{}cpp},
  \texttt{tests/\allowbreak{}cases/\allowbreak{}openfiles.\allowbreak{}expected}}
\def\TCTestTechnique{%
  Back-to-back comparison with recorded reference output.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The sequence of file statements of the Visual Basic 6 experiment
  \texttt{fileTable} of
  \texttt{scripts/\allowbreak{}vb6\_\allowbreak{}harness\_\allowbreak{}gen.\allowbreak{}py};
  \texttt{tests/\allowbreak{}cases/\allowbreak{}openfiles.\allowbreak{}expected}.}
\def\TCInitialState{%
  The working directory \texttt{work-openfiles} of the build tree, used by
  no other test.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}file\_\allowbreak{}tests
      tests/\allowbreak{}cases/\allowbreak{}openfiles.\allowbreak{}expected}.
    \item Run the statement sequence through the helpers, writing the files in
      the working directory.
    \item Record a line for each result (\texttt{label =\textgreater{} value}).
    \item Compare the lines with \texttt{openfiles.\allowbreak{}expected}.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  Every recorded line equals the line the Visual Basic 6 program wrote; the
  program exits 0.}
\def\TCPostConditions{%
  The files the sequence made remain in \texttt{work-openfiles}.}

\TCTemplate

\end{document}

\def\TCID{TC-0016}
\def\TCTitle{%
  Record copy and destroy from a descriptor}
\def\TCPurpose{%
  Verify \texttt{\_\allowbreak{}\_\allowbreak{}vbaRecAssign},
  \texttt{CopyRecord},
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaRecDestruct},
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaRecDestructAnsi}, the ANSI
  conversion, \texttt{\_\allowbreak{}\_\allowbreak{}vbaAryRecCopy} and
  \texttt{\_\allowbreak{}\_\allowbreak{}vbaCVarAryUdt} on descriptors built
  as the Visual Basic 6 compiler builds them.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0005}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}records\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}records.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing of data layouts.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  Descriptors with the size at +2, the field count at +6 and entries from
  +0xc, with dynamic and fixed array fields.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}records\_\allowbreak{}tests}.
    \item Build records and descriptors of each field kind.
    \item Copy, destroy and convert them with the helpers.
    \item Check the copied fields, that Strings and Variants are copied and
      freed, and the locked and chained arrays.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{vbrun records: all checks
  passed} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\def\TCID{TC-0009}
\def\TCTitle{%
  State, strings, arrays, loops, Shell and dialogs}
\def\TCPurpose{%
  Verify the smaller exports: \texttt{Erl} and \texttt{LastDllError}, fixed
  strings, \texttt{Erase}, \texttt{ReDim}, Variant array access,
  \texttt{For} loops of Currency, \texttt{MacID}, the calendar,
  \texttt{Shell}, \texttt{StrConv} with a locale, \texttt{MsgBox},
  \texttt{InputBox}, \texttt{AppActivate} and \texttt{SendKeys}.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003, REQ-0004}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}system\_\allowbreak{}more\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}system.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}scalars.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}varstrings.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest. Under Wine the dialog checks (\texttt{MsgBox},
  \texttt{InputBox}) are relaxed.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  The arguments of each export, including a started child process for
  \texttt{Shell}.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs
      \texttt{vbrun\_\allowbreak{}system\_\allowbreak{}more\_\allowbreak{}tests}.
    \item Run each group (state, strings, arrays, loops, \texttt{Shell},
      dialogs, \texttt{AppActivate}, clock, Variant \texttt{For}, IME,
      \texttt{SendKeys}) and check its results.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass and the program exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}

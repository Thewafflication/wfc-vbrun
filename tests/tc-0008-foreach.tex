\def\TCID{TC-0008}
\def\TCTitle{%
  For Each over arrays, Variants and a collection}
\def\TCPurpose{%
  Verify the \texttt{For Each} helpers over arrays, Variants and a
  collection of the test.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{%
  REQ-0003}
\def\TCDesignRef{%
  Not applicable.}
\def\TCFeatureRef{%
  \texttt{tests/\allowbreak{}vbrun\_\allowbreak{}foreach\_\allowbreak{}tests.\allowbreak{}cpp},
  \texttt{src/\allowbreak{}helpers\_\allowbreak{}foreach.\allowbreak{}cpp}}
\def\TCTestTechnique{%
  Requirements-based testing with a test double.}

\def\TCPreconditions{%
  The library and the test program are built
  (\texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}) and the test
  is registered with CTest.}
\def\TCEnvironment{%
  Windows with the toolchain of
  \texttt{scripts/\allowbreak{}Invoke-Build.\allowbreak{}ps1}; the local
  gate builds and runs x64 Debug, CI builds and runs the other
  architectures. The library is built from this repository in
  \texttt{out/\allowbreak{}build}.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  Arrays of numbers and Strings, the test's collection, an array or an
  object in a Variant, an object without \texttt{\_\allowbreak{}NewEnum}, a
  number, and no array or object.}
\def\TCInitialState{%
  No state from earlier runs is needed; the test makes what it needs.}

\def\TCProcedure{%
  \begin{enumerate}
    \item CTest runs \texttt{vbrun\_\allowbreak{}foreach\_\allowbreak{}tests}.
    \item Start an enumeration with the helpers over each kind of source and
      step through it, with and without \texttt{Exit For}.
    \item Check each element, the state after the last element, that the array
      is locked in the loop and unlocked after it, and the errors: 92 for no
      array, 438 for an object without \texttt{\_\allowbreak{}NewEnum}, 424
      for no object.
    \item CTest records the exit code and, on failure, the program's output
      (\texttt{--output-on-failure}).
  \end{enumerate}}
\def\TCExpectedResult{%
  All checks pass; the program prints \texttt{foreach ok} and exits 0.}
\def\TCPostConditions{%
  No file or registry state remains that later tests depend on.}

\TCTemplate

\end{document}
